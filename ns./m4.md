cat << 'EOF' > server_manager.sh
#!/bin/bash

# ==========================================
# Server Setup, Auto-Monitor & Scheduled Reboot
# ==========================================

# ------------------------------------------
# Telegram Bot Configuration
# ------------------------------------------

BOT_TOKEN="8752327864:AAE7SgGLk345vU9czspxfxVy6VcKuueKNhA"
CHAT_ID="5921429378"

# ------------------------------------------
# File Paths
# ------------------------------------------

MONITOR_SCRIPT="/root/node_monitor.sh"
DAILY_REBOOT_SCRIPT="/root/daily_reboot.sh"
BOOT_NOTIFY_SCRIPT="/root/boot_notify.sh"
LOG_FILE="/var/log/auto_monitor.log"

# Fetch Server IP for Telegram Notifications
SERVER_IP=$(curl -sS --max-time 3 ifconfig.me || echo "Unknown_IP")

# Colors for menu aesthetics
GREEN="\e[32m"
RED="\e[31m"
CYAN="\e[36m"
YELLOW="\e[33m"
RESET="\e[0m"

# Check root access
if [ "$EUID" -ne 0 ]; then
    echo -e "${RED}Error: This script must be run as root!${RESET}"
    exit 1
fi

# Function to send Telegram messages from the main script
send_telegram_main() {
    local MESSAGE="$1"
    curl -s -X POST "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
        -d chat_id="${CHAT_ID}" \
        -d text="${MESSAGE}" \
        -d parse_mode="HTML" > /dev/null 2>&1
}

# ------------------------------------------
# 1. Monitor Script (Every 5 minutes)
# ------------------------------------------

create_monitor_script() {
    cat << 'INNER_EOF' > "$MONITOR_SCRIPT"
#!/bin/bash
BOT_TOKEN="8752327864:AAE7SgGLk345vU9czspxfxVy6VcKuueKNhA"
CHAT_ID="5921429378"
MAX_CPU=90
MAX_RAM=90
LOG_FILE="/var/log/auto_monitor.log"
SERVER_IP=$(curl -sS --max-time 3 ifconfig.me || echo "Unknown_IP")

send_telegram() {
    local MESSAGE="$1"
    curl -s -X POST "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
        -d chat_id="${CHAT_ID}" \
        -d text="${MESSAGE}" \
        -d parse_mode="HTML" > /dev/null 2>&1
}

# Check server uptime (Prevent boot-loop if server just started)
UPTIME_SEC=$(cat /proc/uptime | awk '{print int($1)}')
if [ "$UPTIME_SEC" -lt 600 ]; then
    # If uptime is less than 10 minutes, do nothing
    exit 0
fi

# Calculate CPU and RAM usage
CPU_IDLE=$(vmstat 1 2 | tail -1 | awk '{print $15}')
CPU_USAGE=$((100 - CPU_IDLE))
RAM_USAGE=$(free | grep Mem | awk '{print int($3/$2 * 100.0)}')

if [ "$CPU_USAGE" -ge "$MAX_CPU" ] || [ "$RAM_USAGE" -ge "$MAX_RAM" ]; then
    REASON=""
    if [ "$CPU_USAGE" -ge "$MAX_CPU" ] && [ "$RAM_USAGE" -ge "$MAX_RAM" ]; then
        REASON="مصرف بحرانی CPU ($CPU_USAGE%) و RAM ($RAM_USAGE%)"
    elif [ "$CPU_USAGE" -ge "$MAX_CPU" ]; then
        REASON="مصرف بحرانی CPU ($CPU_USAGE%)"
    else
        REASON="مصرف بحرانی RAM ($RAM_USAGE%)"
    fi

    MSG="🚨 <b>هشدار ریبوت خودکار سرور</b> 🚨
    
🖥 آی‌پی سرور: ${SERVER_IP}
⚠️ دلیل ریستارت: ${REASON}
⏳ وضعیت: سرور برای جلوگیری از هنگی هم‌اکنون ریبوت شد."

    send_telegram "$MSG"

    echo "$(date '+%Y-%m-%d %H:%M:%S') - CRITICAL: CPU:${CPU_USAGE}%, RAM: ${RAM_USAGE}%. Rebooting..." >> "$LOG_FILE"
    sync
    /sbin/reboot
fi
INNER_EOF
    chmod +x "$MONITOR_SCRIPT"
}

# ------------------------------------------
# 2. Daily Reboot Script (At 5:00 AM)
# ------------------------------------------

create_daily_reboot_script() {
    cat << 'INNER_EOF' > "$DAILY_REBOOT_SCRIPT"
#!/bin/bash
BOT_TOKEN="8752327864:AAE7SgGLk345vU9czspxfxVy6VcKuueKNhA"
CHAT_ID="5921429378"
SERVER_IP=$(curl -sS --max-time 3 ifconfig.me || echo "Unknown_IP")

MSG="🔄 ریبوت برنامه‌ریزی شده (روزانه) 🔄

🖥 آی‌پی سرور: ${SERVER_IP}
⏰ زمان اجرا: ۵ صبح
⚙️ وضعیت: سرور جهت آزادسازی منابع و جلوگیری از هنگی، در حال ریستارت شدن است..."

curl -s -X POST "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
    -d chat_id="${CHAT_ID}" \
    -d text="${MSG}" \
    -d parse_mode="HTML" > /dev/null 2>&1

sync
/sbin/reboot
INNER_EOF
    chmod +x "$DAILY_REBOOT_SCRIPT"
}

# ------------------------------------------
# 3. Boot Notify Script (Runs on Startup)
# ------------------------------------------

create_boot_notify_script() {
    cat << 'INNER_EOF' > "$BOOT_NOTIFY_SCRIPT"
#!/bin/bash

# Wait 30 seconds to ensure network interface is fully up
sleep 30

BOT_TOKEN="8752327864:AAE7SgGLk345vU9czspxfxVy6VcKuueKNhA"
CHAT_ID="5921429378"
SERVER_IP=$(curl -sS --max-time 5 ifconfig.me || echo "Unknown_IP")

MSG="✅ سرور با موفقیت در دسترس قرار گرفت ✅

🖥 آی‌پی سرور: ${SERVER_IP}
🟢 وضعیت: سرور روشن شد، ریستارت به درستی انجام گردید و هم‌اکنون شبکه کاملاً متصل و در دسترس است."

curl -s -X POST "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
    -d chat_id="${CHAT_ID}" \
    -d text="${MSG}" \
    -d parse_mode="HTML" > /dev/null 2>&1
INNER_EOF
    chmod +x "$BOOT_NOTIFY_SCRIPT"
}

# ------------------------------------------
# Add all to Cronjob
# ------------------------------------------

setup_cronjobs() {
    # Remove old cronjobs related to this script to avoid duplicates
    crontab -l 2>/dev/null | grep -v -E "$MONITOR_SCRIPT|$DAILY_REBOOT_SCRIPT|$BOOT_NOTIFY_SCRIPT" > /tmp/cron_temp
    
    # Add new jobs
    echo "*/5 * * * * $MONITOR_SCRIPT" >> /tmp/cron_temp
    echo "0 5 * * * $DAILY_REBOOT_SCRIPT" >> /tmp/cron_temp
    echo "@reboot $BOOT_NOTIFY_SCRIPT" >> /tmp/cron_temp
    
    crontab /tmp/cron_temp
    rm -f /tmp/cron_temp
}

# ------------------------------------------
# Core Menu Functions
# ------------------------------------------

install_and_setup() {
    echo -e "${CYAN}Setting server timezone to Asia/Tehran...${RESET}"
    timedatectl set-timezone Asia/Tehran

    echo -e "${CYAN}Updating and upgrading system packages (this may take a few minutes)...${RESET}"
    apt-get update -y && apt-get upgrade -y

    echo -e "${CYAN}Creating monitoring and reboot scripts...${RESET}"
    create_monitor_script
    create_daily_reboot_script
    create_boot_notify_script
    setup_cronjobs

    # Send Setup Notification
    MSG_SETUP="🛠 <b>عملیات نصب اولیه انجام شد</b> 🛠
    
🖥 آی‌پی سرور: ${SERVER_IP}
⚙️ وضعیت: آپدیت سیستم، تنظیم ساعت، فعال‌سازی مانیتورینگ منابع و ریبوت روزانه (۵ صبح) با موفقیت روی این سرور کانفیگ شد. سرور در حال ریبوت است..."
    send_telegram_main "$MSG_SETUP"

    echo -e "${GREEN}Installation and setup completed successfully!${RESET}"
    echo -e "${YELLOW}Server will reboot in 10 seconds...${RESET}"
    sleep 10
    reboot
}

enable_monitoring() {
    create_monitor_script
    create_daily_reboot_script
    create_boot_notify_script
    setup_cronjobs

    # Send Enable Notification
    MSG_ENABLE="🛡 <b>سرویس مانیتورینگ و زمان‌بندی فعال شد</b> 🛡
    
🖥 آی‌پی سرور: ${SERVER_IP}
✅ وضعیت: چک کردن رم و پردازنده (هر ۵ دقیقه) و ریبوت روزانه روی این سرور روشن شد."
    send_telegram_main "$MSG_ENABLE"

    echo -e "${GREEN}Auto-monitoring and Scheduled Reboot successfully [ENABLED].${RESET}"
    sleep 2
}

disable_monitoring() {
    # Remove from cronjob
    crontab -l 2>/dev/null | grep -v -E "$MONITOR_SCRIPT|$DAILY_REBOOT_SCRIPT|$BOOT_NOTIFY_SCRIPT" > /tmp/cron_temp
    crontab /tmp/cron_temp
    rm -f /tmp/cron_temp

    # Remove script files
    rm -f "$MONITOR_SCRIPT" "$DAILY_REBOOT_SCRIPT" "$BOOT_NOTIFY_SCRIPT"

    # Send Disable Notification
    MSG_DISABLE="🛑 <b>سرویس مانیتورینگ غیرفعال شد</b> 🛑
    
🖥 آی‌پی سرور: ${SERVER_IP}
❌ وضعیت: سیستم مانیتورینگ خودکار منابع و ریبوت روزانه متوقف شد."
    send_telegram_main "$MSG_DISABLE"

    echo -e "${RED}Auto-monitoring and Scheduled Reboot successfully [DISABLED].${RESET}"
    sleep 2
}

# ------------------------------------------
# Main Menu
# ------------------------------------------

while true; do
    clear
    echo -e "${CYAN}==========================================${RESET}"
    echo -e "${GREEN}    Server Manager & Auto-Monitor Tool${RESET}"
    echo -e "${CYAN}==========================================${RESET}"
    echo -e " 1) ${YELLOW}Install Setup & Enable Auto-Monitor${RESET}"
    echo -e " 2) ${GREEN}Enable Auto-Monitor Only${RESET}"
    echo -e " 3) ${RED}Disable Auto-Monitor${RESET}"
    echo -e " 0) ${CYAN}Exit${RESET}"
    echo -e "${CYAN}==========================================${RESET}"
    read -p "Select an option [0-3]: " option
    
    case $option in
        1) install_and_setup ;;
        2) enable_monitoring ;;
        3) disable_monitoring ;;
        0) echo -e "${CYAN}Exiting... Have a good day!${RESET}"; exit 0 ;;
        *) echo -e "${RED}Invalid option! Please try again.${RESET}"; sleep 2 ;;
    esac
done
EOF
chmod +x server_manager.sh
bash server_manager.sh
