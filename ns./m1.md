#!/bin/bash

STREAMING_CHUNK:Setting up global variables and colors...

==========================================

Server Setup & Auto-Monitor Script

==========================================

Colors for menu aesthetics

GREEN="\e[32m"
RED="\e[31m"
YELLOW="\e[33m"
CYAN="\e[36m"
RESET="\e[0m"

Paths for monitoring script and logs in the server

MONITOR_SCRIPT="/usr/local/bin/auto_monitor.sh"
LOG_FILE="/var/log/auto_monitor.log"

STREAMING_CHUNK:Adding root privilege check...

Check root access

if [ "$EUID" -ne 0 ]; then
echo -e "${RED}Error: This script must be run as root!${RESET}"
exit 1
fi

STREAMING_CHUNK:Creating the monitor script function...

------------------------------------------

Function to create the monitoring script

------------------------------------------

create_monitor_script() {
cat << 'EOF' > "$MONITOR_SCRIPT"
#!/bin/bash

MAX_CPU=90
MAX_RAM=90
LOG_FILE="/var/log/auto_monitor.log"

Check server uptime (Prevent boot-loop if server just started)

UPTIME_SEC=$(cat /proc/uptime | awk '{print int($1)}')
if [ "$UPTIME_SEC" -lt 600 ]; then
# If uptime is less than 10 minutes, do nothing
exit 0
fi

Calculate CPU and RAM usage

CPU_IDLE=$(vmstat 1 2 | tail -1 | awk '{print $15}')
CPU_USAGE=$((100 - CPU_IDLE))
RAM_USAGE=$(free | grep Mem | awk '{print int($3/$2 * 100.0)}')

if [ "$CPU_USAGE" -ge "$MAX_CPU" ] || [ "$RAM_USAGE" -ge "$MAX_RAM" ]; then
echo "$(date '+\%Y-\%m-\%d \%H:\%M:\%S') - CRITICAL: CPU:${CPU_USAGE}%, RAM: ${RAM_USAGE}%. Rebooting server..." >> "$LOG_FILE"
sync
/sbin/reboot
fi
EOF

chmod +x "$MONITOR_SCRIPT"
}

STREAMING_CHUNK:Defining core functions for setup and cron management...

------------------------------------------

Core Menu Functions

------------------------------------------

install_and_setup() {
echo -e "${CYAN}Setting server timezone to Asia/Tehran...${RESET}"
timedatectl set-timezone Asia/Tehran

echo -e "${CYAN}Updating and upgrading system packages (this may take a few minutes)...${RESET}"
apt-get update -y && apt-get upgrade -y

echo -e "${CYAN}Creating monitoring script...${RESET}"
create_monitor_script

# Add to cronjob (Runs every 5 minutes)
(crontab -l 2>/dev/null | grep -v "$MONITOR_SCRIPT"; echo "*/5 * * * * $MONITOR_SCRIPT") | crontab -

echo -e "${GREEN}Installation and setup completed successfully!${RESET}"
echo -e "${YELLOW}Server will reboot in 10 seconds...${RESET}"
sleep 10
reboot


}

enable_monitoring() {
if [ ! -f "$MONITOR_SCRIPT" ]; then
create_monitor_script
fi
(crontab -l 2>/dev/null | grep -v "$MONITOR_SCRIPT"; echo "*/5 * * * * $MONITOR_SCRIPT") | crontab -
echo -e "${GREEN}Auto-monitoring and reboot successfully [ENABLED].${RESET}"
sleep 2
}

disable_monitoring() {
crontab -l 2>/dev/null | grep -v "$MONITOR_SCRIPT" | crontab -
echo -e "${YELLOW}Auto-monitoring and reboot successfully [DISABLED].${RESET}"
sleep 2
}

STREAMING_CHUNK:Building the interactive menu UI...

------------------------------------------

Interactive Menu

------------------------------------------

show_menu() {
clear
echo -e "${CYAN}================================================${RESET}"
echo -e "${GREEN}  Server Management & Resource Monitoring Tool${RESET}"
echo -e "${CYAN}================================================${RESET}"
echo -e "Current Server Time: $(date '+%Y-%m-%d %H:%M:%S')"
echo -e "------------------------------------------------"
echo -e "${YELLOW}1)${RESET} Initial Setup (Timezone, Update, Install Monitor, Reboot)"
echo -e "${YELLOW}2)${RESET} Enable Monitoring (Auto-reboot on high CPU/RAM)"
echo -e "${YELLOW}3)${RESET} Disable Monitoring"
echo -e "${YELLOW}0)${RESET} Exit"
echo -e "------------------------------------------------"
read -p "Enter your choice (0-3): " choice

case $choice in
    1)
        install_and_setup
        ;;
    2)
        enable_monitoring
        show_menu
        ;;
    3)
        disable_monitoring
        show_menu
        ;;
    0)
        echo -e "${GREEN}Exiting script.${RESET}"
        exit 0
        ;;
    *)
        echo -e "${RED}Invalid choice! Please try again.${RESET}"
        sleep 2
        show_menu
        ;;
esac


}

STREAMING_CHUNK:Executing the main menu...

Run the menu

show_menu
